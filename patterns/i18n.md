## Philosophy

**Lightweight, runtime-loaded JSON catalogs. No compilation, no complex frameworks.**

### Core Principles

1. **Simple key-value JSON** (no ICU complexity in v0.1)
2. **Runtime loading** (no compilation step)
3. **Variable substitution** (`{name}`, `{amount}`)
4. **Fallback to English** (graceful degradation)
5. **Module-specific catalogs** (optional, per-feature translations)

---

## Architecture

### File Structure

```
lang/
  en.json          # English (default/fallback)
  es.json          # Spanish
  fr.json          # French
  ar.json          # Arabic (RTL)
  # ... more languages
```

### JSON Format

```json
{
  "welcome": "Welcome, {name}",
  "errors.E_NOT_FOUND": "Resource not found",
  "errors.E_UNAUTHORIZED": "Please sign in to continue",
  "profile.title": "Profile",
  "profile.updated": "Profile updated successfully",
  "settings.list.title": "Settings",
  "settings.save.success": "Setting saved",
  "settings.delete.confirm": "Are you sure you want to delete this setting?"
}
```

**Namespacing convention:**
- `errors.*` - Error messages
- `<feature>.list.*` - List view strings
- `<feature>.create.*` - Create form strings
- `<feature>.edit.*` - Edit form strings
- `nav.*` - Navigation strings
- `common.*` - Shared strings (save, cancel, delete, etc.)

## Implementation

### Core i18n Package

```go
// internal/platform/i18n/i18n.go
package i18n

import (
  "encoding/json"
  "fmt"
  "os"
  "path/filepath"
  "strings"
  "sync"
)

var (
  translations = make(map[string]map[string]string) // lang -> key -> value
  mu           sync.RWMutex
  loaded       = make(map[string]bool)
)

// Preload loads all translation files for a language at startup
func Preload(lang string) error {
  if loaded[lang] {
    return nil
  }

  path := filepath.Join("lang", lang+".json")

  // Skip if file doesn't exist (language not supported)
  if _, err := os.Stat(path); os.IsNotExist(err) {
    return nil
  }

  data, err := os.ReadFile(path)
  if err != nil {
    return fmt.Errorf("failed to read %s: %w", path, err)
  }

  var langMap map[string]string
  if err := json.Unmarshal(data, &langMap); err != nil {
    return fmt.Errorf("failed to parse %s: %w", path, err)
  }

  mu.Lock()
  translations[lang] = langMap
  loaded[lang] = true
  mu.Unlock()

  return nil
}

// T translates a key with optional variable substitution
// Usage: T("en", "welcome", "name", "Alice")
func T(lang, key string, vars ...string) string {
  mu.RLock()
  langMap, ok := translations[lang]
  mu.RUnlock()

  if !ok {
    // Fallback to English
    mu.RLock()
    langMap, ok = translations["en"]
    mu.RUnlock()

    if !ok {
      return key // Ultimate fallback: return key
    }
  }

  text, ok := langMap[key]
  if !ok {
    // Try English fallback
    mu.RLock()
    if enMap, exists := translations["en"]; exists {
      if enText, exists := enMap[key]; exists {
        text = enText
        ok = true
      }
    }
    mu.RUnlock()

    if !ok {
      return key // Fallback: return key
    }
  }

  // Variable substitution: {name} -> value
  if len(vars) > 0 && len(vars)%2 == 0 {
    for i := 0; i < len(vars); i += 2 {
      placeholder := "{" + vars[i] + "}"
      text = strings.ReplaceAll(text, placeholder, vars[i+1])
    }
  }

  return text
}

// GetLang extracts language from gin context (set by middleware)
func GetLang(c *gin.Context) string {
  if lang, exists := c.Get("lang"); exists {
    return lang.(string)
  }
  return "en" // Default fallback
}

// GetSupportedLanguages returns list of loaded languages
func GetSupportedLanguages() []string {
  mu.RLock()
  defer mu.RUnlock()

  langs := make([]string, 0, len(loaded))
  for lang := range loaded {
    langs = append(langs, lang)
  }
  return langs
}
```

## Middleware

### Language Detection

```go
// internal/platform/i18n/middleware.go
package i18n

import (
  "net/http"
  "strings"

  "github.com/gin-gonic/gin"
)

// Middleware detects language from query param, cookie, or Accept-Language header
func Middleware() gin.HandlerFunc {
  return func(c *gin.Context) {
    lang := detectLanguage(c)

    // Set in context for downstream handlers
    c.Set("lang", lang)

    // Set cookie for persistence (1 year)
    http.SetCookie(c.Writer, &http.Cookie{
      Name:     "lang",
      Value:    lang,
      Path:     "/",
      MaxAge:   31536000, // 1 year
      HttpOnly: true,
      SameSite: http.SameSiteLaxMode,
    })

    c.Next()
  }
}

func detectLanguage(c *gin.Context) string {
  // Priority: 1) Query param, 2) Cookie, 3) Accept-Language, 4) Default

  // 1. Query param (?lang=es)
  if lang := c.Query("lang"); lang != "" && isSupported(lang) {
    return lang
  }

  // 2. Cookie
  if lang, err := c.Cookie("lang"); err == nil && lang != "" && isSupported(lang) {
    return lang
  }

  // 3. Accept-Language header (simplified parsing)
  if accept := c.GetHeader("Accept-Language"); accept != "" {
    // Parse first language code (e.g., "es-ES,en;q=0.9" -> "es")
    parts := strings.Split(accept, ",")
    if len(parts) > 0 {
      langPart := strings.TrimSpace(parts[0])
      if len(langPart) >= 2 {
        lang := langPart[:2] // Get first 2 chars (language code)
        if isSupported(lang) {
          return lang
        }
      }
    }
  }

  // 4. Default
  return "en"
}

func isSupported(lang string) bool {
  mu.RLock()
  _, exists := loaded[lang]
  mu.RUnlock()
  return exists
}
```

```go
func SetupRouter() *gin.Engine {
  router := gin.Default()

  // Register i18n template functions
  router.SetFuncMap(template.FuncMap{
    // Note: This won't work directly since we need context
    // Better approach: pass translation function in data
  })

  // Load templates AFTER SetFuncMap
  router.LoadHTMLGlob("views/**/*.html")

  return router
}
```

### Better Approach: Pass Translation Helper

```go
// In controller
func ListTasks(c *gin.Context) {
  lang := i18n.GetLang(c)

  // Helper closure
  tr := func(key string, vars ...string) string {
    return i18n.T(lang, key, vars...)
  }

  c.HTML(200, "settings/list.html", gin.H{
    "lang":     lang,
    "t":        tr,
    "settings": settings,
  })
}
```

### Template Usage

```html
<!-- views/settings/list.html -->
<h1>{{ t "settings.list.title" }}</h1>

<p>{{ t "welcome" "name" .user.Name }}</p>

<!-- With variables -->
<p>{{ t "profile.updated" }}</p>
```

### Alternative: Pass lang and use a registered function

```go
// Register global template function
router.SetFuncMap(template.FuncMap{
  "t": i18n.T, // Direct function reference
})

// In controller
c.HTML(200, "settings/list.html", gin.H{
  "lang":     lang,
  "settings": settings,
})
```

```html
<!-- In template -->
<h1>{{ t .lang "settings.list.title" }}</h1>
<p>{{ t .lang "welcome" "name" .user.Name }}</p>
```

## Language Detection

### Query Parameter

Change language via URL:
```
https://yourapp.com/dashboard?lang=es
```

Middleware will:
1. Set language in context
2. Store in cookie (persists across requests)

### Cookie

Once set, language persists:
```
Cookie: lang=es
```

### Accept-Language Header

Browser sends preferred language:
```
Accept-Language: es-ES,es;q=0.9,en;q=0.8
```

Middleware parses and selects best match.

## Translation Management

### File Organization

```
lang/
  en.json          # Master (always complete)
  es.json          # Spanish translations
  fr.json          # French translations
  de.json          # German translations
```

### Translation Workflow

1. **Add new keys to en.json** (master)
2. **Copy to other language files**
3. **Translate values** (keep keys identical)
4. **Validate** (ensure all keys exist)

### Validation Script

```bash
#!/bin/bash
# scripts/validate-i18n.sh

MASTER="lang/en.json"
MASTER_KEYS=$(jq -r 'keys[]' "$MASTER" | sort)

for file in lang/*.json; do
  if [ "$file" = "$MASTER" ]; then
    continue
  fi

  echo "Checking $file..."
  FILE_KEYS=$(jq -r 'keys[]' "$file" | sort)

  # Missing keys
  MISSING=$(comm -23 <(echo "$MASTER_KEYS") <(echo "$FILE_KEYS"))
  if [ -n "$MISSING" ]; then
    echo "  Missing keys:"
    echo "$MISSING" | sed 's/^/    /'
  fi

  # Extra keys (in file but not in master)
  EXTRA=$(comm -13 <(echo "$MASTER_KEYS") <(echo "$FILE_KEYS"))
  if [ -n "$EXTRA" ]; then
    echo "  Extra keys (not in master):"
    echo "$EXTRA" | sed 's/^/    /'
  fi
done
```

## RTL Support

### Detect RTL Languages

```go
// internal/platform/i18n/i18n.go

var rtlLanguages = map[string]bool{
  "ar": true, // Arabic
  "fa": true, // Persian/Farsi
  "he": true, // Hebrew
  "ur": true, // Urdu
}

func IsRTL(lang string) bool {
  return rtlLanguages[lang]
}
```

### Middleware Sets Direction

```go
func Middleware() gin.HandlerFunc {
  return func(c *gin.Context) {
    lang := detectLanguage(c)

    c.Set("lang", lang)
    c.Set("isRTL", IsRTL(lang))

    // ... set cookie ...

    c.Next()
  }
}
```

### Base Layout

```html
<!-- views/layouts/base.html -->
<!DOCTYPE html>
<html lang="{{ .lang }}" dir="{{ if .isRTL }}rtl{{ else }}ltr{{ end }}">
<head>
  <!-- ... -->
</head>
<body>
  <!-- Bootstrap 5.3+ automatically flips layout for RTL -->
</body>
</html>
```

### RTL CSS (if needed)

```css
/* static/css/app.css */

/* RTL-specific overrides */
[dir="rtl"] .navbar-brand {
  margin-left: auto;
  margin-right: 0;
}

[dir="rtl"] .text-right {
  text-align: left !important;
}

  text-align: right !important;
```

## Best Practices

### Do's ✅

- ✅ **Namespace keys** (`tasks.list.title`, not just `title`)
- ✅ **Use English as master** (complete all keys in en.json first)
- ✅ **Preload at startup** (don't load on every request)
- ✅ **Fallback gracefully** (lang → en → key)
- ✅ **Keep translations simple** (complex logic in code, not strings)
- ✅ **Validate files** (ensure all languages have same keys)
- ✅ **Use variables** (`{name}`, `{amount}`) for dynamic content
- ✅ **Test RTL** (if supporting Arabic, Hebrew, etc.)

### Don'ts ❌

- ❌ **Don't hardcode strings** in templates or code
- ❌ **Don't use different keys** across languages
- ❌ **Don't translate keys** (keys are identifiers, not text)
- ❌ **Don't put HTML** in translations (use template logic)
- ❌ **Don't handle pluralization yet** (v0.1 - use separate keys)
- ❌ **Don't use complex ICU rules** (keep it simple for now)

## Testing i18n

```go
func TestI18n_Translation(t *testing.T) {
  // Load test translations
  i18n.Preload("en")
  i18n.Preload("es")

  tests := []struct {
    lang string
    key  string
    vars []string
    want string
  }{
    {"en", "welcome", []string{"name", "Alice"}, "Welcome, Alice"},
    {"es", "welcome", []string{"name", "Alice"}, "Bienvenida, Alice"},
    {"en", "nonexistent", nil, "nonexistent"}, // Fallback
  }

  for _, tt := range tests {
    t.Run(tt.lang+"_"+tt.key, func(t *testing.T) {
      got := i18n.T(tt.lang, tt.key, tt.vars...)
      assert.Equal(t, tt.want, got)
    })
  }
}
```
